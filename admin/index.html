Streamer.bot

Create an action and add Execute C# Code. Paste the code below and replace YOUR_SNACK_VAULT_API_KEY with the Bot API key shown above.

using System;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;

public class CPHInline
{
    public bool Execute()
    {
        return ExecuteAsync().GetAwaiter().GetResult();
    }

    private async Task<bool> ExecuteAsync()
    {
        if (!CPH.TryGetArg("userName", out string username) ||
            string.IsNullOrWhiteSpace(username))
        {
            CPH.LogError("Snack Vault: Could not find userName.");
            return false;
        }

        string url = "https://inftkwxwqkyhzrqxmiey.supabase.co/functions/v1/pull";
        string apiKey = "YOUR_SNACK_VAULT_API_KEY";

        string safeUsername = username
            .Replace("\\", "\\\\")
            .Replace("\"", "\\\"");

        string json =
            "{\"username\":\"" + safeUsername +
            "\",\"display_name\":\"" + safeUsername + "\"}";

        try
        {
            using (HttpClient client = new HttpClient())
            {
                client.DefaultRequestHeaders.Add("x-snack-api-key", apiKey);
                client.DefaultRequestHeaders.Add("x-response-format", "text");

                using (StringContent content = new StringContent(
                    json, Encoding.UTF8, "application/json"))
                {
                    HttpResponseMessage response = await client.PostAsync(url, content);
                    string result = await response.Content.ReadAsStringAsync();

                    if (!response.IsSuccessStatusCode)
                    {
                        CPH.LogError("Snack Vault API Error (" +
                            (int)response.StatusCode + "): " + result);
                        return false;
                    }

                    if (string.IsNullOrWhiteSpace(result))
                    {
                        CPH.LogError("Snack Vault returned an empty response.");
                        return false;
                    }

                    CPH.SendMessage(result);
                    return true;
                }
            }
        }
        catch (Exception ex)
        {
            CPH.LogError("Snack Vault Exception: " + ex.Message);
            return false;
        }
    }
}

Attach this action to the Twitch Channel Point reward. Snack Vault sends the pull result directly to Twitch chat.
